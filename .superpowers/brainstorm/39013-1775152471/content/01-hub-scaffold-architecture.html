How it works — Entity Adapter Pattern

destination.adapter.ts
export const destinationAdapter = {
  heroData: (dest) => ({
    image: dest.heroImageUrl,
    badge: 'DESTINATION',
    title: dest.name,
    subtitle: dest.countryName,
  }),

  contextPills: (dest, counts) => [
    { label: `${counts.cruises} Cruises` },
    { label: `${counts.flights} Flights` },
    { label: `${counts.hotels} Hotels` },
    { label: `${counts.offers} Offers`,
      accent: true },
  ],

  // Personalized section order
  sections: (signals) => [
    signals.prefersCruises
      ? 'cruises' : 'flights',
    'hotels',
    'tours',
    'activities',
    'photos',
    'nearby',
    'offers',
  ],
}
ship.adapter.ts
export const shipAdapter = {
  heroData: (ship) => ({
    image: ship.imageUrl,
    badge: 'SHIP',
    title: ship.name,
    subtitle: ship.cruiseLine,
  }),

  contextPills: (ship, counts) => [
    { label: `${counts.sailings} Sailings` },
    { label: `${ship.tonnage} GT` },
    { label: `${ship.passengerCapacity} pax` },
    { label: `Built ${ship.yearBuilt}` },
  ],

  // Ship-specific sections
  sections: () => [
    'sailings',
    'cabinCategories',
    'deckPlans',
    'destinations',
    'shipGallery',
    'moreByCruiseLine',
  ],
}
